```python
def professionals_open(app) -> bool:
    """専門職が現れてよい日付かを返す。自分用の起動は日付を動かせないので、切り替えない。"""
    return app.state.deployment.mine or clock.now(app).date() >= config.PROFESSIONAL_FROM


def can_log_in(user: User | None, app) -> bool:
    if user is None or user.removed_at is not None or user.is_deceased:
        return False
    return user.role is not Role.PROFESSIONAL or professionals_open(app)
```

`current_user`、`login`、`list_users` はこの関数で判定する。`GET /api/me` は `can_invite_professional` を返す。

```python
class ProfessionalInvite(BaseModel):
    model_config = ConfigDict(extra="forbid")
    name: str = Field(min_length=1, max_length=40)
    called_name: str | None = Field(default=None, max_length=40)
    relation: str | None = Field(default=None, max_length=40)


@router.post("/professional", status_code=201)
def invite_professional(body, request, viewer=Depends(current_user), conn=Depends(get_conn)) -> dict:
    """専門職を招く。人の登録と託す設定を1つの処理で入れる。"""
    if not (consent.can_act_for_self(conn, viewer) and professionals_open(request.app)):
        raise HTTPException(status_code=404, detail=NOT_FOUND)
    draft = PersonDraft(name=body.name, called_name=body.called_name, role=Role.PROFESSIONAL, relation=body.relation)
    with ledger.atomic(conn):
        person = add_person(conn, draft)
        conn.execute(
            "INSERT INTO delegations (delegate_user_id, scope, decided_on) VALUES (?, ?, ?)",
            (person.id, ShareScope.DECISION_SUPPORT.value, clock.now(request.app).date().isoformat()),
        )
    return person_view(person)


@router.get("/choices")
def list_choices(request, viewer=Depends(current_user), conn=Depends(get_conn)) -> dict:
    """部屋の相手の一覧。生まれた年やメモは返さない。"""
    people = [user for user in db.login_users(conn) if user.id != viewer.id and can_log_in(user, request.app)]
    return {"items": [person_view(user) for user in people]}
```

`add_person` は `with conn:` を使うので、外側の処理の中で動くよう `ledger.atomic` に変える。

```python
def find_pair_room(conn: sqlite3.Connection, first: int, second: int) -> Room | None:
    """2人とも入っている1対1の部屋を返す。"""
    row = conn.execute(
        """
        SELECT r.* FROM rooms r
        WHERE r.kind = ?
          AND EXISTS (SELECT 1 FROM room_members a WHERE a.room_id = r.id AND a.user_id = ? AND a.left_at IS NULL)
          AND EXISTS (SELECT 1 FROM room_members b WHERE b.room_id = r.id AND b.user_id = ? AND b.left_at IS NULL)
        """,
        (RoomKind.PAIR.value, first, second),
    ).fetchone()
    return db.row_to_model(row, Room) if row else None
```

```python
def creator_of(conn: sqlite3.Connection, room_id: int) -> int | None:
    """部屋を作った人を返す。作るときに、作った人を先頭で入れている。"""
    row = conn.execute(
        "SELECT user_id FROM room_members WHERE room_id = ? ORDER BY rowid LIMIT 1", (room_id,)
    ).fetchone()
    return row["user_id"] if row else None


def remove_member(conn: sqlite3.Connection, room_id: int, user_id: int, moment: str) -> None:
    conn.execute(
        "UPDATE room_members SET left_at = ? WHERE room_id = ? AND user_id = ? AND left_at IS NULL",
        (moment, room_id, user_id),
    )
```

```sql
CREATE TABLE IF NOT EXISTS room_reads (
    room_id INTEGER NOT NULL REFERENCES rooms(id),
    user_id INTEGER NOT NULL REFERENCES users(id),
    last_read_seq INTEGER NOT NULL,
    updated_at TEXT NOT NULL,
    PRIMARY KEY (room_id, user_id)
);
```

部屋の行は追記だけなので、部屋の中の順番（`seq`）は変わらない。順番で持つと、画面の番号と同じ単位で比べられる。
